% Decoding-temperature ablation, DeepSeek V4 Pro, false-presupposition setting.
% 4 cells x 100 items = 400 runs, T = 25, naturalistic proxy (Claude Sonnet 5,
% thinking on, full-window memory), judge Claude Sonnet 5, wind-down stop off.
% Only the target's sampling temperature varies across rows.
%
% Requires: booktabs.
% ---------------------------------------------------------------------------
\begin{table}[t]
  \centering
  \small
  \begin{tabular}{lccccc}
    \toprule
    & \multicolumn{5}{c}{False Presupposition} \\
    \cmidrule(l){2-6}
    \textbf{Target temperature} & $n$ & CR@5 & CR@25 & TH@25 & AUSC \\
    \midrule
    $\tau = 0.0$          & 100 & 49\% & 86\% & 7.8 & 0.22 \\
    $\tau = 0.3$          & 100 & 46\% & 90\% & 8.0 & 0.23 \\
    $\tau = 0.6$ (default)& 100 & 49\% & 88\% & 7.7 & 0.23 \\
    $\tau = 1.0$          & 100 & 48\% & 85\% & 8.5 & 0.24 \\
    \midrule
    \multicolumn{1}{l}{\emph{spread}} & --- & 3\,pt & 5\,pt & 0.8 & 0.02 \\
    \bottomrule
  \end{tabular}
  \caption{Decoding temperature does not affect sycophantic collapse.
    DeepSeek V4 Pro under the naturalistic proxy at $T=25$, with the target's
    sampling temperature the only variable ($400$ runs total; every other
    setting, including proxy, judge, turn budget and random item order, held
    fixed). No metric separates the four cells: collapse rate spans
    $85$--$90\%$, and the ordering is not monotonic in $\tau$ --- $\tau=0.3$ is
    the most collapse-prone cell and $\tau=1.0$ the least. A $\chi^{2}$ test
    across the four collapse counts gives $p=0.72$; a Kruskal--Wallis test on
    the collapse-turn distributions gives $p=0.77$. The benchmark therefore
    measures a property of the model rather than an artifact of the decoding
    configuration, and the $\tau=0.6$ operating point used throughout the paper
    is not a favourable choice.}
  \label{tab:temp-ablation}
\end{table}
